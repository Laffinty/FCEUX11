// header
class XxxDialog_t : public QDialog
{
    Q_OBJECT
public:
    XxxDialog_t(QWidget *parent = 0);
protected:
    void changeEvent(QEvent* event) override;  // ADD
private:
    void retranslateUi();                       // ADD
    QPushButton* okBtn;   // 改为成员指针（非构造期传 tr()）
    // ... 其他 UI 元素同理
};

// cpp
void XxxDialog_t::changeEvent(QEvent* event)
{
    if (event->type() == QEvent::LanguageChange) {
        retranslateUi();
    }
    QDialog::changeEvent(event);
}

void XxxDialog_t::retranslateUi()
{
    setWindowTitle(tr("..."));
    if (okBtn) okBtn->setText(tr("&OK"));
    // ... 所有构造期 new QPushButton(tr(...), this) 改为成员指针 + setText
}
```

**关键纪律**：构造期 `new QPushButton(tr("..."), this)` 必须改为成员指针 + `setText(tr(...))`，否则运行期无法重设。

// ConfigStore.h
#pragma once
#include <QSettings>
#include <QString>
#include <optional>
#include <type_traits>

namespace fceu11::qt {

/// Compile-time typed wrapper around QSettings. The key path is
/// generated from a static string literal; the value type is checked
/// at compile time. This replaces the 74 hand-rolled
/// 'QSettings().value("foo/bar", default).toBool()' calls scattered
/// across src/drivers/Qt/.
///
/// Iron rule: TypedConfig instances are constructed lazily on first
/// use and cached as statics; do NOT construct one per-call.
template <typename T>
class TypedConfig {
public:
    TypedConfig(const char* key, T defaultValue)
        : m_key(key), m_default(defaultValue) {}

    T get() const {
        QSettings s;
        return readImpl(s);
    }
    void set(const T& v) const {
        QSettings s;
        writeImpl(s, v);
    }
    const char* key() const { return m_key; }
private:
    T readImpl(QSettings& s) const;
    void writeImpl(QSettings& s, const T& v) const;
    const char* m_key;
    T m_default;
};

} // namespace
```
